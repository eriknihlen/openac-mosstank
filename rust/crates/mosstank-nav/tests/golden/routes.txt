# Route scenarios for the black-box golden run (RustPortNavGolden.cs writes
# routes.out from the C# controller; tests/golden.rs replays them in Rust).
#
# Setup lines: route MODE MIN [max= portal= pathing= doors= identify= open=
# follow= corners= enabled=]; wp TYPE [at=EW,NS[,EL] ref=EW,NS id= name=
# text= dur= recall= spell= spellname= heading= shift= charge= dir=];
# me EW NS HEADING [air] [portal] [gone] [confirmed=EW,NS,REV];
# obj ID NAME EW,NS [door] [open] [locked] [lockstate] [diff=N]; unobj ID;
# find ID NAME EW,NS; nofind; typing 0|1; vendor ID; appraisal 0|1;
# magic [answer=] [casting=] [completion=REV,SPELL,ERR];
# usecompletion REV SOURCE ERR; goanswer STATUS; goend STATE REASON;
# chat SEQ LOGTYPE SENDERID KIND "sender" "text" "display"; locks;
# lockadvance S. A number ending in m is metres (divided by 240). A
# double-quoted word may hold spaces (no quotes inside it).
#
# Action lines (each prints the calls made and the state after): tick S
# [noact]; frame S; claim [noact]; step S [blocked]; steps S; door S
# [noact]; macrostop; lostturn; reset; anchor; skip N; back N; reverse;
# clearlocks; pathing MODE; warnreset; query.
